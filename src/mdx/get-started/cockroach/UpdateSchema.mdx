Add a `phone` column to the `users` table in `src/db/schema.ts`.

```typescript copy filename="src/db/schema.ts" {10}
import { cockroachTable, uniqueIndex, int4, varchar } from "drizzle-orm/cockroach-core"

export const users = cockroachTable(
  'users',
  {
    id: int4().primaryKey().generatedAlwaysAsIdentity(),
    name: varchar({ length: 255 }).notNull(),
    age: int4().notNull(),
    email: varchar({ length: 255 }).notNull(),
    phone: varchar(),
  },
  (table) => [
    uniqueIndex('users_email_unique').using('btree', table.email.asc()),
  ],
);
```
